  <!-- Theme toggle wiring -->
  <script>
    (function () {
      function current() { return document.documentElement.getAttribute('data-theme') === 'dark' ? 'dark' : 'light'; }
      function apply(theme) {
        document.documentElement.setAttribute('data-theme', theme);
        try { localStorage.setItem('s2v_theme', theme); } catch (e) {}
        window.dispatchEvent(new CustomEvent('s2v:themechange', { detail: { theme: theme } }));
      }
      function syncControls() {
        var theme = current();
        document.querySelectorAll('[data-theme-set]').forEach(function (btn) {
          btn.setAttribute('aria-pressed', btn.getAttribute('data-theme-set') === theme ? 'true' : 'false');
        });
      }
      window.S2V_toggleTheme = function () { apply(current() === 'dark' ? 'light' : 'dark'); syncControls(); };
      document.addEventListener('click', function (e) {
        var explicit = e.target.closest('[data-theme-set]');
        if (explicit) { apply(explicit.getAttribute('data-theme-set')); syncControls(); return; }
        if (e.target.closest('[data-theme-toggle]')) window.S2V_toggleTheme();
      });
      document.addEventListener('DOMContentLoaded', syncControls);
      window.addEventListener('s2v:themechange', syncControls);
    })();
  </script>
